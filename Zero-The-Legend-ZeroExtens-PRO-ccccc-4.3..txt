    class Entity {

    class Reader {
        constructor(buffer) {
            this.dataView = new DataView(buffer);
            this.byteOffset = 0;
            this.bufferLength = buffer.byteLength;
        }
        remaining() { return this.bufferLength - this.byteOffset; }
        hasMore() { return this.byteOffset < this.bufferLength; }
        readUint8() {
            if (this.remaining() < 1) return 0;
            return this.dataView.getUint8(this.byteOffset++);
        }
        readUint16() {
            if (this.remaining() < 2) return 0;
            const v = this.dataView.getUint16(this.byteOffset, true);
            this.byteOffset += 2;
            return v;
        }
        readInt32() {
            if (this.remaining() < 4) return 0;
            const v = this.dataView.getInt32(this.byteOffset, true);
            this.byteOffset += 4;
            return v;
        }
        readUint32() {
            if (this.remaining() < 4) return 0;
            const v = this.dataView.getUint32(this.byteOffset, true);
            this.byteOffset += 4;
            return v;
        }
        readFloat64() {
            if (this.remaining() < 8) return 0;
            const v = this.dataView.getFloat64(this.byteOffset, true);
            this.byteOffset += 8;
            return v;
        }
        readString() {
            let result = "";
            while (this.byteOffset < this.bufferLength) {
                const c = this.dataView.getUint8(this.byteOffset++);
                if (c === 0) break;
                result += String.fromCharCode(c);
            }
            return result;
        }
        readUTF8String() {
            const bytes = [];
            while (this.byteOffset < this.bufferLength) {
                const b = this.dataView.getUint8(this.byteOffset++);
                if (b === 0) break;
                bytes.push(b);
            }
            return new TextDecoder("utf-8").decode(new Uint8Array(bytes));
        }
    }

    class Writer {
        constructor(size) {
            this.dataView = new DataView(new ArrayBuffer(size || 1000));
            this.byteOffset = 0;
        }
        ensureCapacity(extra) {
            let cap = this.dataView.buffer.byteLength;
            if (this.byteOffset + extra <= cap) return;
            while (this.byteOffset + extra > cap) cap *= 2;
            const grown = new ArrayBuffer(cap);
            new Uint8Array(grown).set(new Uint8Array(this.dataView.buffer));
            this.dataView = new DataView(grown);
        }
        writeUint8(v) { this.ensureCapacity(1); this.dataView.setUint8(this.byteOffset++, v); }
        writeInt32(v) { this.ensureCapacity(4); this.dataView.setInt32(this.byteOffset, v, true); this.byteOffset += 4; }
        writeUint32(v) { this.ensureCapacity(4); this.dataView.setUint32(this.byteOffset, v, true); this.byteOffset += 4; }
        writeUTF8String(str) {
            const bytes = new TextEncoder().encode(str);
            this.ensureCapacity(bytes.length + 1);
            for (let i = 0; i < bytes.length; i++) this.dataView.setUint8(this.byteOffset++, bytes[i]);
            this.dataView.setUint8(this.byteOffset++, 0);
        }
        toBuffer() { return this.dataView.buffer.slice(0, this.byteOffset); }
    }

    const diag = { closes: {}, recycled: {}, resets: 0, op85: 0, deaths: 0 };
    const stormGuard = { recent: [], episodes: [], cooldownUntil: 0 };

    function noteCloseForStorm() {
        const now = Date.now();
        stormGuard.recent.push(now);
        while (stormGuard.recent.length && now - stormGuard.recent[0] > 1000) stormGuard.recent.shift();
        const total = Math.max(bots.length, 1);
        const threshold = Math.max(15, Math.ceil(total * 0.4));
        if (stormGuard.recent.length >= threshold) {
            const lastEpisode = stormGuard.episodes[stormGuard.episodes.length - 1];
            if (!lastEpisode || now - lastEpisode > 1500) stormGuard.episodes.push(now);
        }
        while (stormGuard.episodes.length && now - stormGuard.episodes[0] > 20000) stormGuard.episodes.shift();
        if (stormGuard.episodes.length >= 2) stormGuard.cooldownUntil = now + 8000;
    }

    const connector = {
        queue: [],
        active: new Set(),
        timer: null,
        spacing: 50,
        maxInflight: 3,
        enqueue(bot) {
            if (bot.queued) return;
            bot.queued = true;
            this.queue.push(bot);
            if (!this.timer) this.timer = setInterval(() => this.tick(), this.spacing);
        },
        release(bot) {
            this.active.delete(bot);
        },
        remove(bot) {
            this.active.delete(bot);
            if (!bot.queued) return;
            bot.queued = false;
            const i = this.queue.indexOf(bot);
            if (i !== -1) this.queue.splice(i, 1);
        },
        tick() {
            for (const b of this.active) {
                if (!b.ws || b.ws.readyState !== 0) this.active.delete(b);
            }
            if (this.active.size >= this.maxInflight) return;
            const bot = this.queue.shift();
            if (!bot) {
                clearInterval(this.timer);
                this.timer = null;
                return;
            }
            bot.queued = false;
            this.active.add(bot);
            try { bot.open(); } catch (e) { this.active.delete(bot); }
        }
    };

    class Bot {
        constructor(config, customName = null, index = 0) {
            this.config = config;
            this.ws = null;
            this.serverUrl = null;
            this.stopped = false;
            this.queued = false;
            this.connected = false;
            this.isAlive = false;
            this.moveInt = null;
            this.connectTimeout = null;
            this.errorTimeout = null;
            this.reconnectTimeout = null;
            this.respawnTimeout = null;
            this.isReconnecting = false;
            this.connectionAttempts = 0;
            this.reconnectBase = 400;
            this.reconnectStep = 400;
            this.reconnectMax = 3000;
            this.connectionTimeout = 12000;
            this.handshakeTimeout = 15000;
            this.spawnRetry = 3000;
            this.stallTimeout = 25000;
            this.clientVersion = 31116;
            this.protocolVersion = 23;
            this.biggerPlayerAvoidanceRange = 1.15;
            this.avoidanceDistance = 1;
            this.name = customName || storage.get("gx-bot-tag") || "Zero The Legend";
            this.index = index;
            this.resetState();
            this.requestConnect();
        }

        resetState() {
            this.stopMoveLoop();
            this.isAlive = false;
            this.connected = false;
            this.spawnReady = false;
            this.spawnTries = 0;
            this.lastSpawnAt = 0;
            this.lastMessageAt = Date.now();
            this.stateSince = Date.now();
            this.offsetX = 0;
            this.offsetY = 0;
            this.cells = new Map();
            this.pellets = new Map();
            this.pelletCache = null;
            this.ownSet = new Set();
            this.myCellIDs = [];
            this.encryptionKey = 0;
            this.decryptionKey = 0;
            this.serverVersion = null;
            this.followMouse = this.config.botAi ? false : !this.config.vShield;
            this.fleeing = false;
        }

        clearTimers() {
            for (const k of ["connectTimeout", "errorTimeout", "reconnectTimeout", "respawnTimeout"]) {
                if (this[k]) { clearTimeout(this[k]); this[k] = null; }
            }
            this.isReconnecting = false;
        }

        detachSocket() {
            connector.release(this);
            const ws = this.ws;
            this.ws = null;
            if (ws) {
                ws.onopen = ws.onclose = ws.onerror = ws.onmessage = null;
                try { ws.close(); } catch (e) {}
            }
        }

        requestConnect() {
            this.clearTimers();
            connector.remove(this);
            this.detachSocket();
            this.resetState();
            if (this.stopped || !this.config.startedBots) return;
            connector.enqueue(this);
        }

        open() {
            if (this.stopped || !this.config.startedBots) { connector.release(this); return; }
            if (!this.config.agarServer) { connector.release(this); this.scheduleReconnect(); return; }
            try {
                this.serverUrl = this.config.agarServer;
                const ws = new WebSocket(this.serverUrl);
                ws.__scarz = true;
                ws.binaryType = "arraybuffer";
                ws.onopen = () => this.onopen(ws);
                ws.onclose = (ev) => this.onclose(ws, ev);
                ws.onerror = () => this.onerror(ws);
                ws.onmessage = (e) => this.onmessage(ws, e);
                this.ws = ws;
                this.stateSince = Date.now();
                this.connectionAttempts++;
                this.connectTimeout = setTimeout(() => {
                    if (ws.readyState === WebSocket.CONNECTING) { try { ws.close(); } catch (e) {} }
                }, this.connectionTimeout);
            } catch (e) {
                connector.release(this);
                this.scheduleReconnect();
            }
        }

        scheduleReconnect() {
            if (this.stopped || !this.config.startedBots || this.isReconnecting) return;
            this.isReconnecting = true;
            let delay = Math.min(this.reconnectBase + this.connectionAttempts * this.reconnectStep, this.reconnectMax) + Math.random() * 600;
            const now = Date.now();
            if (now < stormGuard.cooldownUntil) {
                delay += Math.min(stormGuard.cooldownUntil - now, 8000) * 0.5 + Math.random() * 1500;
            }
            this.reconnectTimeout = setTimeout(() => {
                this.isReconnecting = false;
                this.requestConnect();
            }, delay);
        }

        recycle(reason) {
            diag.recycled[reason] = (diag.recycled[reason] || 0) + 1;
            this.clearTimers();
            this.detachSocket();
            this.connected = false;
            this.isAlive = false;
            this.stopMoveLoop();
            this.scheduleReconnect();
        }

        check(now) {
            if (this.stopped || !this.config.startedBots) return;
            const ws = this.ws;
            if (!ws) {
                if (!this.isReconnecting && !this.queued) this.requestConnect();
                return;
            }
            if (ws.readyState === WebSocket.CONNECTING) {
                if (now - this.stateSince > this.connectionTimeout + 2000) this.recycle('connect');
                return;
            }
            if (ws.readyState !== WebSocket.OPEN) {
                if (!this.isReconnecting && now - this.stateSince > 3000) this.recycle('closing');
                return;
            }
            if (this.isAlive) {
                if (now - this.lastMessageAt > this.stallTimeout) this.recycle('stall');
                return;
            }
            if (this.spawnReady) {
                if (now - this.lastSpawnAt > this.spawnRetry) {
                    if (this.spawnTries < 3) {
                        this.spawnTries++;
                        this.sendSpawn();
                    } else {
                        this.recycle('nospawn');
                    }
                }
            } else if (now - this.stateSince > this.handshakeTimeout) {
                this.recycle('handshake');
            }
        }

        onopen(ws) {
            if (ws !== this.ws) return;
            connector.release(this);
            this.connected = true;
            this.stateSince = Date.now();
            this.lastMessageAt = Date.now();
            this.sendProtocolVersion();
            this.sendClientVersion();
        }

        onclose(ws, ev) {
            if (ws !== this.ws) return;
            connector.release(this);
            const code = ev && ev.code;
            diag.closes[code] = (diag.closes[code] || 0) + 1;
            noteCloseForStorm();
            this.connected = false;
            this.isAlive = false;
            this.stateSince = Date.now();
            this.stopMoveLoop();
            this.scheduleReconnect();
        }

        onerror(ws) {
            if (ws !== this.ws) return;
            this.errorTimeout = setTimeout(() => {
                try { ws.close(); } catch (e) {}
            }, 1000);
        }

        onmessage(ws, event) {
            if (ws !== this.ws) return;
            this.lastMessageAt = Date.now();
            let data = event.data;
            if (!(data instanceof ArrayBuffer)) return;
            if (this.decryptionKey) data = this.xorBuffer(data, this.decryptionKey ^ this.clientVersion);
            this.handleBuffer(data);
        }

        send(buffer) {
            const ws = this.ws;
            if (!ws || ws.readyState !== WebSocket.OPEN) return;
            if (this.encryptionKey) {
                buffer = this.xorBuffer(buffer, this.encryptionKey);
                this.encryptionKey = this.rotateKey(this.encryptionKey);
            }
            ws.send(buffer);
        }

        startMoveLoop() {
            this.stopMoveLoop();
            this.moveInt = setInterval(() => this.move(), 40);
        }

        stopMoveLoop() {
            if (this.moveInt) { clearInterval(this.moveInt); this.moveInt = null; }
        }

        closeSoon() {
            const ws = this.ws;
            setTimeout(() => { try { if (ws) ws.close(); } catch (e) {} }, 1000);
        }

        handleBuffer(buffer) {
            try {
                if (!buffer || buffer.byteLength < 1) return;
                const reader = new Reader(buffer);
                const type = reader.readUint8();

                switch (type) {
                    case 18:
                        diag.resets++;
                        this.cells.clear();
                        this.pellets.clear();
                        this.ownSet.clear();
                        this.myCellIDs.length = 0;
                        this.markDead();
                        break;
                    case 85:
                        diag.op85++;
                        this.closeSoon();
                        break;
                    case 32:
                        if (reader.remaining() >= 4) {
                            const id = reader.readUint32();
                            if (!this.myCellIDs.includes(id)) this.myCellIDs.push(id);
                            this.ownSet.add(id);
                        }
                        if (!this.isAlive) {
                            this.isAlive = true;
                            this.connectionAttempts = 0;
                            this.spawnTries = 0;
                            this.stateSince = Date.now();
                            this.startMoveLoop();
                        }
                        break;
                    case 241:
                        if (reader.remaining() >= 4) {
                            this.decryptionKey = reader.readUint32();
                            this.serverVersion = reader.readString();
                            const match = this.serverUrl && this.serverUrl.match(SERVER_RE);
                            if (match) this.encryptionKey = this.murmur2(match[1] + this.serverVersion, 255);
                        }
                        break;
                    case 242:
                        this.spawnReady = true;
                        this.sendSpawn();
                        break;
                    case 255:
                        if (reader.remaining() >= 4) {
                            const outSize = reader.readUint32();
                            if (outSize > 0 && outSize <= 8 * 1024 * 1024) {
                                const out = this.uncompressMessage(new Uint8Array(buffer, 5), new Uint8Array(outSize));
                                this.handleMessage(out.buffer);
                            }
                        }
                        break;
                }
            } catch (e) {}
        }

        handleMessage(buffer) {
            try {
                if (!buffer || buffer.byteLength < 1) return;
                const reader = new Reader(buffer);
                const type = reader.readUint8();
                if (type === 16) this.updateNodes(reader);
                else if (type === 64) this.updateOffset(reader);
            } catch (e) {}
        }

        updateOffset(reader) {
            if (reader.remaining() < 32) return;
            const minX = reader.readFloat64();
            const minY = reader.readFloat64();
            const maxX = reader.readFloat64();
            const maxY = reader.readFloat64();
            if (maxX - minX > 14000) this.offsetX = (maxX + minX) / 2;
            if (maxY - minY > 14000) this.offsetY = (maxY + minY) / 2;
        }

        removeCell(id) {
            this.cells.delete(id);
            this.pellets.delete(id);
            this.ownSet.delete(id);
            const i = this.myCellIDs.indexOf(id);
            if (i !== -1) this.myCellIDs.splice(i, 1);
        }

        markDead() {
            if (!this.isAlive) return;
            diag.deaths++;
            this.isAlive = false;
            this.spawnTries = 0;
            this.stateSince = Date.now();
            this.stopMoveLoop();
            this.followMouse = this.config.botAi ? false : !this.config.vShield;
            this.fleeing = false;
            this.respawnTimeout = setTimeout(() => this.sendSpawn(), 100);
        }

        updateNodes(reader) {
            if (reader.remaining() < 2) return;

            const eatCount = reader.readUint16();
            for (let i = 0; i < eatCount; i++) {
                if (reader.remaining() < 8) return;
                reader.readUint32();
                this.removeCell(reader.readUint32());
            }

            while (reader.remaining() >= 4) {
                const id = reader.readUint32();
                if (id === 0) break;
                if (reader.remaining() < 11) return;

                let entity = this.cells.get(id) || this.pellets.get(id);
                if (!entity) {
                    entity = new Entity();
                    entity.id = id;
                }
                entity.x = reader.readInt32();
                entity.y = reader.readInt32();
                entity.size = reader.readUint16();

                const flags = reader.readUint8();
                const extendedFlags = flags & 128 ? reader.readUint8() : 0;

                if (flags & 1) entity.isVirus = true;
                if (flags & 2) { if (reader.remaining() < 3) return; reader.byteOffset += 3; }
                if (flags & 4) reader.readString();
                if (flags & 8) entity.name = reader.readUTF8String();
                if (extendedFlags & 1) entity.isPellet = true;
                if (extendedFlags & 2) entity.isFriend = true;
                if (extendedFlags & 4) { if (reader.remaining() < 4) return; reader.byteOffset += 4; }
                if (entity.isPellet) this.pellets.set(id, entity);
                else this.cells.set(id, entity);
            }

            if (reader.remaining() >= 2) {
                const removed = reader.readUint16();
                for (let i = 0; i < removed && reader.remaining() >= 4; i++) {
                    this.removeCell(reader.readUint32());
                }
            }

            if (this.isAlive && this.myCellIDs.length === 0) this.markDead();
        }

        nearestPellet(bot, now) {
            const c = this.pelletCache;
            if (c && now - c.at < 4000) {
                const cached = this.pellets.get(c.id);
                if (cached) return cached;
            }
            let best = null, bestD = Infinity;
            for (const e of this.pellets.values()) {
                if (e.isFriend) continue;
                const dx = e.x - bot.x;
                const dy = e.y - bot.y;
                const d = dx * dx + dy * dy;
                if (d < bestD) { bestD = d; best = e; }
            }
            this.pelletCache = best ? { id: best.id, at: now } : null;
            return best;
        }

        scan(bot, now) {
            const r = {
                bigger: null, biggerDist: Infinity,
                smaller: null, smallerDist: Infinity,
                pellet: null,
                virus: null, virusDist: Infinity
            };
            for (const e of this.cells.values()) {
                if (e.isPellet || e.isFriend || this.ownSet.has(e.id)) continue;
                const dx = e.x - bot.x;
                const dy = e.y - bot.y;
                const dist = Math.max(0, Math.sqrt(dx * dx + dy * dy) - (bot.size + e.size));

                if (e.isVirus) {
                    if (bot.size > e.size * 1.32 && dist < r.virusDist) { r.virusDist = dist; r.virus = e; }
                } else if (e.name !== this.name) {
                    if (e.size > bot.size * this.biggerPlayerAvoidanceRange) {
                        if (dist < r.biggerDist) { r.biggerDist = dist; r.bigger = e; }
                    } else if (e.size < bot.size) {
                        if (dist < r.smallerDist) { r.smallerDist = dist; r.smaller = e; }
                    }
                }
            }
            r.pellet = this.nearestPellet(bot, now);
            return r;
        }

        steerAway(bot, e) {
            this.moveTo(bot.x + (bot.x - e.x), bot.y + (bot.y - e.y));
        }

        shouldFlee(enemy, dist, enterRange) {
            if (!enemy) { this.fleeing = false; return false; }
            if (this.fleeing) {
                if (dist > enterRange * 1.5) { this.fleeing = false; return false; }
                return true;
            }
            if (dist < enterRange) { this.fleeing = true; return true; }
            return false;
        }

        randomMove(bot) {
            const randomX = Math.floor(1337 * Math.random());
            const randomY = Math.floor(1337 * Math.random());
            this.moveTo(bot.x + (Math.random() > 0.5 ? randomX : -randomX), bot.y + (Math.random() > 0.5 ? -randomY : randomY));
        }

        farm(bot, s) {
            if (s.pellet) this.moveTo(s.pellet.x, s.pellet.y);
            else this.randomMove(bot);
        }

        move() {
            if (!this.isAlive || this.myCellIDs.length === 0) return;

            let sx = 0, sy = 0, n = 0, size = 0, maxCell = 0;
            for (const id of this.myCellIDs) {
                const c = this.cells.get(id);
                if (!c) continue;
                sx += c.x; sy += c.y; size += c.size; n++;
                if (c.size > maxCell) maxCell = c.size;
            }
            if (!n) return;
            const bot = { x: sx / n, y: sy / n, size };
            const cords = this.config.cords;

            // --- NUOVA FUNZIONE 1: Tricksplit Auto-Eject Routine ---
            if (this.config.tricksplitActive) {
                this.fleeing = false;
                this.moveTo(cords.x + this.offsetX, cords.y + this.offsetY);
                this.split();
                this.eject();
                return;
            }

            // --- NUOVA FUNZIONE 2: Patrol Orbit Mode ---
            if (this.config.patrolMode) {
                this.fleeing = false;
                const angle = (Date.now() / 600) + (this.index * (Math.PI * 2 / Math.max(1, bots.length)));
                const radius = 350;

        uncompressMessage(compressed, output) {
            try {
                for (let i = 0, j = 0; i < compressed.length;) {
                    const token = compressed[i++];
                    let literalLength = token >> 4;
                    if (literalLength > 0) {
                        let ext = literalLength + 240;
                        for (; ext === 255;) {
                            if (i >= compressed.length) return output;
                            ext = compressed[i++];
                            literalLength += ext;
                        }
                        const end = Math.min(i + literalLength, compressed.length);
                        const n = Math.min(end - i, output.length - j);
                        if (n > 0) output.set(compressed.subarray(i, i + n), j);
                        j += end - i;
                        i = end;
                        if (i === compressed.length) return output;
                    }
                    if (i + 1 >= compressed.length) return output;
                    const offset = compressed[i++] | compressed[i++] << 8;
                    if (offset === 0 || offset > j) return output;
                    let matchLength = token & 15;
                    let ext = matchLength + 240;
                    for (; ext === 255;) {
                        if (i >= compressed.length) return output;
                        ext = compressed[i++];
                        matchLength += ext;
                    }
                    let pos = j - offset;
                    const end = Math.min(j + matchLength + 4, output.length);
                    for (; j < end;) output[j++] = output[pos++];
                }
                return output;
            } catch (e) { return output; }
        }

        murmur2(str, seed) {
            let length = str.length;
            let h = seed ^ length;
            let i = 0;
            let k;
            while (length >= 4) {
                k = str.charCodeAt(i) & 255 | (str.charCodeAt(++i) & 255) << 8 | (str.charCodeAt(++i) & 255) << 16 | (str.charCodeAt(++i) & 255) << 24;
                k = (k & 65535) * 1540483477 + (((k >>> 16) * 1540483477 & 65535) << 16);
                k ^= k >>> 24;
                k = (k & 65535) * 1540483477 + (((k >>> 16) * 1540483477 & 65535) << 16);
                h = (h & 65535) * 1540483477 + (((h >>> 16) * 1540483477 & 65535) << 16) ^ k;
                length -= 4;
                ++i;
            }
            switch (length) {
                case 3: h ^= (str.charCodeAt(i + 2) & 255) << 16;
                case 2: h ^= (str.charCodeAt(i + 1) & 255) << 8;
                case 1: h ^= str.charCodeAt(i) & 255;
                    h = (h & 65535) * 1540483477 + (((h >>> 16) * 1540483477 & 65535) << 16);
            }
            h ^= h >>> 13;
            h = (h & 65535) * 1540483477 + (((h >>> 16) * 1540483477 & 65535) << 16);
            h ^= h >>> 15;
            return h >>> 0;
        }

    function setText(el, text) {
        if (el && el.textContent !== text) el.textContent = text;
    }

    function pad2(n) { return String(n).padStart(2, "0"); }

    function render() {
        if (!uiRoot) return;
        const running = botConfig.startedBots;

        const startBtn = $(".gx-stfinish");
        const stopBtn = $(".gx-stop");
        if (startBtn) startBtn.style.display = running ? "none" : "inline-flex";
        if (stopBtn) stopBtn.style.display = running ? "inline-flex" : "none";

        let alive = 0, connected = 0, connecting = 0;
        for (const b of bots) {
            if (b.isAlive) alive++;
            if (b.connected) connected++;
            if (b.ws && b.ws.readyState === 0) connecting++;
        }
        setText($(".gx-botCount"), running ? `${alive}/${botConfig.botCount}` : String(botConfig.botCount));

        // Aggiorna Massa Totale Bot (Nuova Funzione 5)
        setText($(".gx-botMass"), running ? String(getTotalBotMass()) : "0");

        const stat = $(".gx-stat");
        if (stat) {
            const waiting = Math.max(0, bots.length - connected - connecting);
            const stormLeft = Math.max(0, stormGuard.cooldownUntil - Date.now());
            const tip = `alive ${alive} · connected ${connected} · connecting ${connecting} · waiting ${waiting}\ncloses ${JSON.stringify(diag.closes)}\nreset ${diag.resets} · op85 ${diag.op85} · deaths ${diag.deaths}\nrecycled ${JSON.stringify(diag.recycled)}\ncrash-guard ${stormLeft > 0 ? "active (" + Math.ceil(stormLeft / 1000) + "s)" : "idle"}`;
            if (stat.title !== tip) stat.title = tip;
        }

        let text, cls;
        if (notice && Date.now() < notice.until) { text = notice.text; cls = "status-stopped"; }
        else if (running) { text = "Running"; cls = "status-running"; }
        else if (hasRun) { text = "Stopped"; cls = "status-stopped"; }
        else { text = "Online"; cls = "status-green"; }
        setText($("#status"), text);
        const light = $("#status-light");
        if (light) {
            const want = "status-indicator " + cls;
            if (light.className !== want) light.className = want;
        }

        const sw = $("#stopwatch");
        if (sw) {
            sw.style.display = running ? "block" : "none";
            if (running && startTime) {
                const e = Date.now() - startTime;
                setText(sw, `${pad2(Math.floor(e / 60000))}:${pad2(Math.floor((e % 60000) / 1000))}`);
            } else {
                setText(sw, "00:00");
            }
        }

        const ai = $(".gx-ai-btn");
        if (ai) { setText(ai, `AI ${botConfig.botAi ? "ON" : "OFF"}`); ai.classList.toggle("active", botConfig.botAi); }
        const vs = $(".gx-vs-btn");
        if (vs) { setText(vs, `VSH ${botConfig.vShield ? "ON" : "OFF"}`); vs.classList.toggle("active", botConfig.vShield); }

        // Render stato nuovi pulsanti
        const frz = $(".gx-frz-btn");
        if (frz) { setText(frz, `FRZ ${botConfig.freezeBots ? "ON" : "OFF"}`); frz.classList.toggle("active", botConfig.freezeBots); }
        const ptl = $(".gx-ptl-btn");
        if (ptl) { setText(ptl, `PTL ${botConfig.patrolMode ? "ON" : "OFF"}`); ptl.classList.toggle("active", botConfig.patrolMode); }
    }

    function buildUI() {
        const host = document.createElement("div");
        host.id = panelId;
        host.style.cssText = "all:initial;position:fixed;top:0;left:0;width:0;height:0;overflow:visible;" +
            "z-index:2147483647;pointer-events:none;display:block;";
        const root = host.attachShadow({ mode: "open" });
        root.innerHTML = `<style>${CSS}</style>${PANEL_HTML}`;

        const on = (sel, type, fn) => {
            const el = root.querySelector(sel);
            if (el) el.addEventListener(type, fn);
        };
        on(".gx-stfinish", "click", () => startBots("start"));
        on(".gx-stop", "click", () => startBots("stop"));
        on(".gx-ai-btn", "click", toggleAI);
        on(".gx-vs-btn", "click", toggleVShield);
        on(".gx-frz-btn", "click", toggleFreeze);
        on(".gx-ptl-btn", "click", togglePatrol);
        on(".cfg-btn", "click", toggleSettings);

        root.querySelectorAll("button").forEach((b) => {
            b.tabIndex = -1;
            b.addEventListener("mousedown", (e) => e.preventDefault());
        });
        ["mousedown", "mouseup", "dblclick", "contextmenu", "wheel"].forEach((t) => {
            root.addEventListener(t, (e) => e.stopPropagation());
        });

        uiHost = host;
        uiRoot = root;
        document.documentElement.appendChild(host);
    }

    function saveSettings() {
        const nameEl = $("#botName");
        const amountEl = $("#gx-bot-qty");
        const targetNameEl = $("#targetPlayerName");
        if (!nameEl || !amountEl) return;

        const botName = nameEl.value.trim() || "XEVBOTS";
        let amount = parseInt(amountEl.value);
        if (isNaN(amount) || amount < 1) amount = 10;
        if (amount > 500) amount = 500;
        storage.set("gx-bot-tag", botName);
        storage.set("gx-bot-qty", String(amount));
        botConfig.botCount = amount;

        if (targetNameEl) {
            botConfig.targetPlayerName = targetNameEl.value.trim();
            botConfig.clonePlayerName = botConfig.targetPlayerName.length > 0;
        }

        bots.forEach((b, idx) => { b.name = `${botName}${idx + 1}`; });

        const key = (id, fallback) => {
            const el = $(id);
            return (el && el.value.trim().toUpperCase()) || fallback;
        };
        botConfig.keybinds.splitKey = key("#key-split", "X");
        botConfig.keybinds.feedKey = key("#key-feed", "C");
        botConfig.keybinds.modeKey = key("#key-mode", "F");
        botConfig.keybinds.vShieldKey = key("#key-vshield", "V");
        botConfig.keybinds.freezeKey = key("#key-freeze", "T");
        botConfig.keybinds.patrolKey = key("#key-patrol", "P");
        botConfig.keybinds.trickKey = key("#key-trick", "G");
        storage.set("gx-keybinds", JSON.stringify(botConfig.keybinds));

        updateBotCount();
        render();
    }

    function toggleSettings() {
        const modal = $("#settings-modal");
        if (modal) {
            saveSettings();
            modal.remove();
        } else {
            openSettings();
        }

    function loadKeybinds() {
        const saved = storage.get("gx-keybinds");
        if (!saved) return;
        try {
            const p = JSON.parse(saved);
            const k = botConfig.keybinds;
            k.modeKey = p.modeKey || k.modeKey;
            k.feedKey = p.feedKey || k.feedKey;
            k.splitKey = p.splitKey || k.splitKey;
            k.vShieldKey = p.vShieldKey || k.vShieldKey;
            k.freezeKey = p.freezeKey || "T";
            k.patrolKey = p.patrolKey || "P";
            k.trickKey = p.trickKey || "G";
        } catch (e) {}
    }

    function isTyping(event) {
        const path = typeof event.composedPath === "function" ? event.composedPath() : [];
        const t = path[0] || event.target;
        if (!t || !t.tagName) return false;
        const tag = t.tagName;
        return tag === "INPUT" || tag === "TEXTAREA" || tag === "SELECT" || t.isContentEditable === true;
    }

    function matchKey(event, bind) {
        if (!bind) return false;
        const b = bind.toUpperCase();
        if (event.key && event.key.length === 1 && event.key.toUpperCase() === b) return true;
        return event.code === "Key" + b || event.code === "Digit" + b;
    }

    function onKeyDown(event) {
        if (event.ctrlKey || event.metaKey || event.altKey) return;
        if (isTyping(event)) return;
        const kb = botConfig.keybinds;

        if (event.key === "F1") {
            event.preventDefault();
            startBots("start");
            return;
        }
        if (event.key === "F2") {
            event.preventDefault();
            startBots("stop");
            return;
        }
        if (event.key === "F3") {
            event.preventDefault();
            toggleSettings();
            return;
        }

        if (matchKey(event, kb.modeKey)) {
            event.preventDefault();
            if (!event.repeat) toggleAI();
        } else if (matchKey(event, kb.feedKey)) {
            event.preventDefault();
            if (bots.length > 0 && !botConfig.botAi) bots.forEach((b) => { try { b.eject(); } catch (e) {} });
        } else if (matchKey(event, kb.splitKey)) {
            event.preventDefault();
            if (bots.length > 0 && !botConfig.botAi) bots.forEach((b) => { try { b.split(); } catch (e) {} });
        } else if (matchKey(event, kb.vShieldKey)) {
            event.preventDefault();
            if (!event.repeat) toggleVShield();
        } else if (matchKey(event, kb.freezeKey)) {
            event.preventDefault();
            if (!event.repeat) toggleFreeze();
        } else if (matchKey(event, kb.patrolKey)) {
            event.preventDefault();
            if (!event.repeat) togglePatrol();
        } else if (matchKey(event, kb.trickKey)) {
            event.preventDefault();
            if (!event.repeat) triggerTricksplit();
        }
    }

    function onMouseMove(event) {
        botConfig.cords._rawX = event.clientX;
        botConfig.cords._rawY = event.clientY;
        try {
            if (window.__scarzBots && window.__scarzBots.setMouse) window.__scarzBots.setMouse(event.clientX, event.clientY);
        } catch (e) {}
    }

    let boundListeners = [];

    function attachGlobalListeners() {
        boundListeners.forEach(([target, type, fn, opts]) => {
            try { target.removeEventListener(type, fn, opts); } catch (e) {}
        });
        boundListeners = [];
        const add = (target, type, fn, opts) => {
            target.addEventListener(type, fn, opts);
            boundListeners.push([target, type, fn, opts]);
        };
        add(window, "keydown", onKeyDown, true);
        add(window, "mousemove", onMouseMove, true);
    }

    function hookWebSocket() {
        const blocked = ["delt.io", "ixagar", "glitch", "socket.io", "firebase", "agartool.io", "herokuapp", "localhost"];
        const isBlocked = (url) => blocked.some((d) => url.includes(d));
        const proto = WebSocket.prototype;
        const original = proto.send;
        proto.send = function () {
            if (!this.__scarz) {
                try {
                    const url = this.url;
                    if (url && !isBlocked(url) && /agar/i.test(url)) botConfig.agarServer = url;
                } catch (e) {}
            }
            return original.apply(this, arguments);
        };
    }

    function ensureUI() {
        if (document.readyState === "loading" || !document.documentElement || !document.body) return;
        if (!window.__scarzBots) injectDeltaScript();
        if (uiHost && uiHost.isConnected) return;
        buildUI();
        attachGlobalListeners();
        render();
    }

    let ensureScheduled = false;
    function scheduleEnsure() {
        if (ensureScheduled) return;
        ensureScheduled = true;
        setTimeout(() => { ensureScheduled = false; ensureUI(); }, 120);
    }

    window._gxStart = startBots;
    window._gxAI = toggleAI;
    window._gxVS = toggleVShield;
    window._gxCfg = toggleSettings;

    if (/agar\.io/.test(location.hostname) || /delt\.io/.test(location.hostname)) {
        loadKeybinds();
        hookWebSocket();
        setInterval(updateMouseWorld, 33);
        setInterval(render, 250);
        setInterval(ensureUI, 1000);
        try {
            new MutationObserver(scheduleEnsure).observe(document, { childList: true, subtree: true });
        } catch (e) {}
        ensureUI();
    }
